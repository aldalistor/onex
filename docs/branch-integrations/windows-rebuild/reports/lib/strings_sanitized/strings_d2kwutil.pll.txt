PROCEDURE StartFile ( DocumentFile IN VARCHAR2,
vcExtension VARCHAR2(30);
vcClass VARCHAR2(256);
vcCmd VARCHAR2(1000);
Win_API_Debug.DebugLog(99,'StartFile','Args: DocumentFile="'||DocumentFile||'", WindowStyle='||to_char(WindowStyle)||', RaiseExceptions='||Win_API_Debug.BoolToChar(RaiseExceptions));
vcExtension := substr(DocumentFile,instr(DocumentFile,'.',-1));
vcClass := WIN_API_ENVIRONMENT.Read_Registry('HKEY_CLASSES_ROOT\'||vcExtension, null, FALSE);
if vcClass is NULL then /* not registered type */
win_api.add_error('StartFile','Files of Type "'||vcExtension||'" do not have an associated application.');
RAISE NO_DATA_FOUND;
vcCmd := WIN_API_ENVIRONMENT.Read_Registry( 'HKEY_CLASSES_ROOT\'||vcClass||'\shell\open\command', null, FALSE);
if vcCmd is NULL then /* This type of file cannot be opened */
win_api.add_error('StartFile','Files of Class "'||vcClass||'" do not have an associated shell open command.');
if instr(vcCmd, '%1') > 0 then
vcCmd := replace(vcCmd, '%1', DocumentFile);
vcCmd := vcCmd||' "'||DocumentFile||'"';
WIN_API_SHELL.WinExec( vcCmd,WindowStyle,TRUE);
RAISE;
end StartFile;

PROCEDURE Register_Function_Call ( FunctionID in PLS_INTEGER,
FunctionName in VARCHAR2,
FunctionReturn in VARCHAR2,
FunctionArgs in VARCHAR2) is
vcArgs VARCHAR2(200);
vcFrag VARCHAR2(10);
iOffset PLS_INTEGER;
-- See if the DLL has been loaded at all.
-- if this call to FIND_LIBRARY fails, we'll fall through to the Exception.
DLL_HANDLE := ORA_FFI.find_library(DLL_NAME);
when ora_ffi.ffi_error then
-- Attempt to load the DLL again in a subblock. If this fails
-- then the API must be disabled and NOT_DATA_FOUND raised
DLL_HANDLE := Attempt_DLL_Load(DLL_Path, DLL_NAME);
API_ENABLED := FALSE;
add_error('D2KWUTIL',DLL_NAME||' not found.');
-- now compare the Cached value for this function's loadcount with
-- that for th
e DLL this tells us that the function has either not
-- been registered yet, or that the DLL has been reloaded in between
-- times. This would not be healthy so we have to re-register in that case.
if (not FunctionLoad.exists(FunctionId)) or (FunctionLoad(FunctionId)<> DLL_LOAD_COUNT) then
-- We need to go through the registration process.
-- Register the function with the DLL
FunctionHandle(FunctionId) := ora_ffi.register_function(Win_API.DLL_HANDLE,FunctionName,ORA_FFI.C_STD);
if FunctionName <> 'd2kwutil_DebugString' then
if not FunctionLoad.exists(FunctionId) then
Win_API_Debug.DebugLog(99,'Register_Function_Call','Ftn="'||FunctionName||'", DLL Count='||to_char(DLL_LOAD_COUNT)||', Initial Reg');
Win_API_Debug.DebugLog(99,'Register_Function_Call','Ftn="'||FunctionName||'", DLL Count='||to_char(DLL_LOAD_COUNT)||', Re-Registration Count = '||to_char(FunctionLoad(FunctionId)));
-- Loop through and register the parameters
vcArgs := FunctionArgs||',';
while vcArgs is not null and vcArgs <> ',' and vcArgs <> 'VOID,' LOOP
iOffset := instr(vcArgs,',');
vcFrag := substr(vcArgs,1,iOffset-1);
ora_ffi.register_parameter(FunctionHandle(FunctionId),TranslateType(vcFrag));
if iOffset = length(vcArgs) then
vcArgs := NULL;
vcArgs := substr(vcArgs,iOffset+1);
end LOOP;
-- Register the return type IF this is a function
if upper(FunctionReturn) <> 'VOID' and FunctionReturn is not NULL then
ora_ffi.register_return(FunctionHandle(FunctionId),TranslateType(FunctionReturn));
-- Now add this Function's Loadcount to the array to prevent un-nessary re-registration
FunctionLoad(FunctionId) := DLL_LOAD_COUNT;
Win_API.Add_error(FunctionName, Win_API.UError||WIN_API.Add_External_errors);
end Register_Function_Call;
FUNCTION Attempt_DLL_Load (TargetPath in out VARCHAR2, TargetDLL in VARCHAR2, LoadCount in BOOLEAN DEFAULT TRUE) return ORA_FFI.LibHandleType is

hFile TEXT_IO.FILE_TYPE;

-- Watch out for folks who put multiple Paths in here
-- Oh yes it's happened! Just use the first & throw the rest away
-- Add trailing Directory Sep if required
TargetPath := StripDirSep(TargetPath)||'\';
-- Check that there is actually a file there.
hFile := TEXT_IO.FOPEN(TargetPath||TargetDLL,'R');
TEXT_IO.FCLOSE(hFile);
Win_API_Debug.DebugLog(99,'Attempt_DLL_LOAD',TargetDLL||' not found in the specified location of "'||TargetPath||'"');
raise BadLocation;
hDLL := ORA_FFI.LOAD_LIBRARY(TargetPath, TargetDLL);
if LoadCount then
DLL_LOAD_COUNT := DLL_LOAD_COUNT + 1;
API_ENABLED := TRUE;
return hDLL;
return hDLL; /* Will be null */
end Attempt_DLL_Load;

iRC := ORA_FFI.C_LONG_PTR;
elsif vcType = 'LONG' then
iRC := ORA_FFI.C_LONG;
elsif vcType = '*SHORT' or vcType = '*BYTE' then

iRC := ORA_FFI.C_SHORT;
elsif vcType = '*DOUBLE' then
iRC := ORA_FFI.C_DOUBLE_PTR;
elsif vcType = 'DOUBLE' then
iRC := ORA_FFI.C_DOUBLE;

end TranslateType;
